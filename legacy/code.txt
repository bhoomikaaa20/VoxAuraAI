import streamlit as st
from dotenv import load_dotenv
import os
import requests
import shelve
from pathlib import Path
import time
import tempfile
from langchain.embeddings import OpenAIEmbeddings
from langchain.llms import OpenAI
from langchain.document_loaders import TextLoader
from langchain.vectorstores import Chroma
from langchain.chains import RetrievalQA
from langchain.chat_models import ChatOpenAI  # Import ChatOpenAI for chat-based models

load_dotenv()

api_token = os.getenv("assemblyai_api_key")
openai_api_key = os.getenv("openai_api_key")
os.environ['OPENAI_API_KEY'] = openai_api_key

# AssemblyAI base URL and headers
base_url = "https://api.assemblyai.com/v2"
headers = {
    "authorization": api_token,
    "content-type": "application/json"
}

USER_AVATAR = "👤"
BOT_AVATAR = "🤖"

# Function to transcribe audio
def assembly_trans(audio_filename):
    try:
        # Upload audio file to AssemblyAI
        with open(audio_filename, "rb") as f:
            response = requests.post(base_url + '/upload', headers=headers, data=f)
        response.raise_for_status()
        upload_url = response.json()['upload_url']

        # Request transcription
        data = {
            "audio_url": upload_url
        }
        url = base_url + '/transcript'
        response = requests.post(url, json=data, headers=headers)
        response.raise_for_status()
        transcript_id = response.json()['id']

        # Polling for transcription completion
        polling_endpoint = f"https://api.assemblyai.com/v2/transcript/{transcript_id}"
        while True:
            transcript_result = requests.get(polling_endpoint, headers=headers)
            transcript_result.raise_for_status()
            transcript_json = transcript_result.json()
            if transcript_json['status'] == 'completed':
                break
            elif transcript_json['status'] == 'error':
                raise RuntimeError(f"Transcription failed: {transcript_json['error']}")
            else:
                time.sleep(3)

        transcription_text = transcript_json['text']
        os.makedirs('docs', exist_ok=True)
        with open('docs/transcription.txt', 'w') as file:
            file.write(transcription_text)

        return transcription_text

    except Exception as e:
        st.error(f"Error in transcription: {e}")
        return None

@st.cache_resource
def get_vectorstore():
    try:
        loader = TextLoader('docs/transcription.txt')
        documents = loader.load()
        embeddings = OpenAIEmbeddings()
        temp_dir = tempfile.mkdtemp()
        st.session_state['persist_directory'] = temp_dir
        vectorstore = Chroma.from_documents(documents, embeddings, persist_directory=temp_dir)
        return vectorstore
    except Exception as e:
        st.error(f"Error initializing vectorstore: {e}")
        return None

# Efficient LLM using GPT-3.5-Turbo for cost efficiency
def langchain_QA(query, retriever):
    try:
        # Using GPT-3.5-turbo for better cost and performance
        llm = ChatOpenAI(model="gpt-3.5-turbo", temperature=0)  # Use ChatOpenAI for chat models
        qa_chain = RetrievalQA.from_chain_type(
            llm=llm,
            chain_type="stuff",
            retriever=retriever
        )
        result = qa_chain.run(query)
        return result
    except Exception as e:
        st.error(f"Error in langchain_QA: {e}")
        return None

# Load chat history from shelve file
def load_chat_history():
    with shelve.open("chat_history") as db:
        return db.get("messages", [])

# Save chat history to shelve file
def save_chat_history(messages):
    with shelve.open("chat_history") as db:
        db["messages"] = messages

# Streamlit app configuration
st.set_page_config(layout='centered', page_title='VoxAuraAI', page_icon='🔉')
st.title("VoxAuraAI")

# Input for the audio file
uploaded_audio = st.file_uploader("Upload an audio file (MP3, WAV, etc.)", type=["mp3", "wav", "m4a"])

# Initialize chat history
if "messages" not in st.session_state:
    st.session_state.messages = load_chat_history()

# Sidebar with delete history button
with st.sidebar:
    if st.button("Delete Chat History"):
        st.session_state.messages = []
        save_chat_history([])

# Display chat messages
for message in st.session_state.messages:
    avatar = USER_AVATAR if message["role"] == "user" else BOT_AVATAR
    with st.chat_message(message["role"], avatar=avatar):
        st.markdown(message["content"])

# Audio processing logic
if uploaded_audio:
    st.info("Your uploaded audio file")
    st.audio(uploaded_audio)

    # Save the uploaded file temporarily
    temp_audio_path = tempfile.NamedTemporaryFile(delete=False, suffix=".mp3").name
    with open(temp_audio_path, "wb") as f:
        f.write(uploaded_audio.read())

    # Transcribe the audio
    transcription = assembly_trans(temp_audio_path)
    st.write(transcription)
    if transcription:
        st.success("Transcription completed.")

        # Initialize vector store and retriever
        vectorstore = get_vectorstore()
        if vectorstore is not None:
            retriever = vectorstore.as_retriever()
            st.session_state['retriever'] = retriever

            # Main chat interface for query input
            if prompt := st.chat_input("Ask a new question here"):
                # Append user message to chat history
                st.session_state.messages.append({"role": "user", "content": prompt})
                with st.chat_message("user", avatar=USER_AVATAR):
                    st.markdown(prompt)

                # Get response from the vectorstore retriever
                with st.chat_message("assistant", avatar=BOT_AVATAR):
                    message_placeholder = st.empty()
                    full_response = ""
                    result = langchain_QA(prompt, retriever)
                    full_response += result if result else "No answer found."
                    message_placeholder.markdown(full_response)

                # Append assistant response to chat history
                st.session_state.messages.append({"role": "assistant", "content": full_response})

                # Save chat history after each interaction
                save_chat_history(st.session_state.messages)
